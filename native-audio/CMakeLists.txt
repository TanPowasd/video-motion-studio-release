cmake_minimum_required(VERSION 3.25)
project(vmotion_audio VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
set(SMTG_USE_STATIC_CRT ON CACHE BOOL "" FORCE)
set(SMTG_ENABLE_VST3_PLUGIN_EXAMPLES OFF CACHE BOOL "" FORCE)
set(SMTG_ENABLE_VST3_HOSTING_EXAMPLES OFF CACHE BOOL "" FORCE)
set(SMTG_ENABLE_VSTGUI_SUPPORT OFF CACHE BOOL "" FORCE)
set(SMTG_RUN_VST_VALIDATOR OFF CACHE BOOL "" FORCE)
set(SMTG_CREATE_PLUGIN_LINK OFF CACHE BOOL "" FORCE)
add_subdirectory(${VMOTION_VST3_SDK} sdk)
add_executable(vmotion-audio main.cpp vst-host.cpp
 ${VMOTION_VST3_SDK}/public.sdk/source/vst/hosting/plugprovider.cpp
 ${VMOTION_VST3_SDK}/public.sdk/source/common/memorystream.cpp)
target_include_directories(vmotion-audio PRIVATE ${VMOTION_VST3_SDK} ${VMOTION_JSON_INCLUDE})
target_link_libraries(vmotion-audio PRIVATE sdk_hosting)
if(WIN32)
 target_compile_definitions(vmotion-audio PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
 target_link_libraries(vmotion-audio PRIVATE ole32 user32)
 # SDK 3.8.1 loadAsPackage uses the locale-dependent path(string) constructor.
 # Build a local UTF-8 correction without mutating the pinned upstream checkout.
 file(READ ${VMOTION_VST3_SDK}/public.sdk/source/vst/hosting/module_win32.cpp VMOTION_MODULE_WIN32)
 string(REPLACE "filesystem::path p (inPath);" "filesystem::path p = filesystem::u8path (inPath);" VMOTION_MODULE_WIN32 "${VMOTION_MODULE_WIN32}")
 file(WRITE ${CMAKE_CURRENT_BINARY_DIR}/module_win32_utf8.cpp "${VMOTION_MODULE_WIN32}")
 target_include_directories(vmotion-audio PRIVATE ${VMOTION_VST3_SDK}/public.sdk/source/vst/hosting)
 target_sources(vmotion-audio PRIVATE ${CMAKE_CURRENT_BINARY_DIR}/module_win32_utf8.cpp)
elseif(APPLE)
 enable_language(OBJCXX)
 target_sources(vmotion-audio PRIVATE au-host.mm ${VMOTION_VST3_SDK}/public.sdk/source/vst/hosting/module_mac.mm)
 target_link_libraries(vmotion-audio PRIVATE "-framework Cocoa" "-framework AudioToolbox" "-framework AudioUnit" "-framework CoreAudio")
else()
 target_sources(vmotion-audio PRIVATE ${VMOTION_VST3_SDK}/public.sdk/source/vst/hosting/module_linux.cpp)
 target_link_libraries(vmotion-audio PRIVATE dl pthread)
endif()
smtg_add_vst3plugin(vmotion-audio-fixture fixture.cpp
 ${VMOTION_VST3_SDK}/public.sdk/source/vst/vstsinglecomponenteffect.cpp)
target_link_libraries(vmotion-audio-fixture PRIVATE sdk)
target_include_directories(vmotion-audio-fixture PRIVATE ${VMOTION_VST3_SDK})
if(WIN32)
 target_link_libraries(vmotion-audio-fixture PRIVATE user32)
 target_compile_definitions(vmotion-audio-fixture PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
endif()
